+// code_boss follows its channel's tag (check-update), and takes whatever build is there when it is not the one it runs -
+// an older one too. So promoting a build, or putting an old one back, is moving that tag: nothing is built.

+/** The remote a push goes to: the repository with the login in it, for that one push only - the stored remote (and
+ * .git/config) never has the token. A folder, ssh or other non-http source as it is. */
+export function setupAuthedRemote(updateSource, creds) {
+  if (!/^https?:/i.test(updateSource)) return updateSource;
+  const u = new URL(updateSource);
+  u.username = creds.user;
+  u.password = creds.token;
+  if (!u.pathname.endsWith('.git')) u.pathname = u.pathname.replace(/\/+$/, '') + '.git';
+  return u.href;

+/**
+ * The releases in the release folder, newest first: [{ sha, build, at, channels: [...] }], and where each channel is.
+ * { ok: false, problem } when the folder is not a release folder yet.
+ */
+export function relChannelList(dir, { max = 40 } = {}) {
+  if (!dir || !existsSync(path.join(dir, '.git'))) return { ok: false, problem: `There is no release folder at ${dir || '(none set)'} yet - its first release makes it.` };
+  const log = rcGit(dir, ['log', 'main', `-n${max}`, '--format=%H%x09%cI%x09%s']);
+  if (log.status !== 0) return { ok: false, problem: 'Its history could not be read: ' + rcHide(log.stderr || log.stdout).trim().slice(0, 300) };
+  const at = {};
+  const refs = rcGit(dir, ['show-ref', '--tags']);
+  for (const l of String(refs.stdout || '').split(/\r?\n/)) {
+    const m = /^([0-9a-f]{40}) refs\/tags\/(\S+)$/.exec(l.trim());
+    if (m && REL_CHANNELS.includes(m[2])) at[m[2]] = m[1];
+  }

+    const [sha, when, ...subject] = l.split('\t');
+    const m = /Release build (\S+)/.exec(subject.join('\t'));
+    return { sha, at: when, build: m ? m[1] : '', channels: REL_CHANNELS.filter((c) => at[c] === sha) };
+  });
+  // a channel on a commit older than the ones listed: said, not lost
+  const elsewhere = REL_CHANNELS.filter((c) => at[c] && !builds.some((b) => b.sha === at[c]));
+  return { ok: true, builds, channels: at, elsewhere };
