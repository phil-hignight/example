+// ── THE RELEASE'S OWN FOLDERS: never part of the project, even when they sit inside its folder ─────────────────
+/** Where the release builds (the setup file's extract): the saved one, else code_boss-src beside the checkout. */
+export const rsReleaseDir = (checkout) => rsReadJson(rsCfgPath()).releaseDir || path.join(path.dirname(path.resolve(checkout)), 'code_boss-src');
+/** The dist folder (release-channels.mjs): the saved one, else code_boss-dist beside the release folder. */
+export const rsDistDir = (checkout) => String(rsReadJson(rsCfgPath()).distDir || '').trim() || path.join(path.dirname(rsReleaseDir(checkout)), 'code_boss-dist');
+/**
+ * THOSE OF THEM INSIDE THE CHECKOUT, as git names them ('code_boss-src'), [] when none. Prod, 2026-11-16: the release
+ * folder was D:\code\code_boss\code_boss-project\code_boss-src, inside the project folder; the setup window's `git add -A`
+ * had put its files in the project's git, so every release - which rewrites that folder - left 25 "changed" files, and
+ * the next release was refused for them ("This folder gets its files from chunks, and 25 of them are not committed").
+ */
+export function rsFoldersInside(checkout) {
+  if (!checkout) return [];
+  const out = [];
+  for (const d of [rsReleaseDir(checkout), rsDistDir(checkout)]) {
+    const rel = path.relative(path.resolve(checkout), path.resolve(d));
+    if (!rel || rel.startsWith('..') || path.isAbsolute(rel)) continue;
+    const p = rel.split(path.sep).join('/');
+    if (!out.some((o) => o.toLowerCase() === p.toLowerCase())) out.push(p);
+  }
+  return out;
+}
+/** A path as git names it, in one of those folders? */
+export const rsUnderFolders = (p, dirs) => {
+  const l = String(p).replace(/^"|"$/g, '').toLowerCase();
+  return dirs.some((d) => { const x = d.toLowerCase(); return l === x || l.startsWith(x + '/'); });
+};
+/**
+ * Those folders taken out of the checkout's git (git rm --cached: the files stay where they are) and kept out
+ * (.git/info/exclude, so a `git add -A` does not put them back). git: (args) => a spawnSync result in the checkout.
+ * The folders that were in git's list. Throws when git cannot take one out.
+ */
+export function rsKeepFoldersOut(checkout, git) {
+  const dirs = rsFoldersInside(checkout), dropped = [];
+  for (const d of dirs) {
+    const ls = git(['ls-files', '-z', '--', d]);
+    if (ls.status !== 0 || !String(ls.stdout || '').length) continue;
+    const r = git(['rm', '-r', '--cached', '-q', '--', d]);
+    if (r.status !== 0) throw new Error(`git could not take ${d} out of its list: ${String(r.stderr || r.stdout || '').trim().slice(0, 300)}`);
+    dropped.push(d);
+  }
+  if (dirs.length) {
+    const ex = path.join(checkout, '.git', 'info', 'exclude');
+    const t = existsSync(ex) ? readFileSync(ex, 'utf8') : '';
+    const have = t.split(/\r?\n/);
+    const add = dirs.map((d) => `/${d}/`).filter((l) => !have.includes(l));
+    if (add.length) { mkdirSync(path.dirname(ex), { recursive: true }); writeFileSync(ex, t + (t && !t.endsWith('\n') ? '\n' : '') + add.join('\n') + '\n'); }
+  }
+  return dropped;
+}
