+//
+// Every call from every window asks before it goes - `await scheduler.admit({ level, model, tokens, who, server,
+// signal })`, awaited in the caller's own async context (the scheduler only resolves a promise; it never runs the
+// caller's work from its own timers, so AsyncLocalStorage contexts survive) - and says what it used when it ends
+// (`ticket.release(usage)`).

+//   level 2  sub-agents (the review checker included)
+//   level 3  Heads up, the dev check, the skills maintainer, recall, commit suggestions - the rest

+// PRIORITIZE, AND DON'T HAMMER THE SERVICE AFTER A 429 (the developer, 2026-11-20: "If there are messages waiting we
+// just use priority (w/ the building pressure for lower priorities) and send as many as we can until 429s then wait the
+// requested amount of time"):
+// - a call goes at once, whatever its level, unless calls to its model are waiting;
+// - a 429 holds the model for the wait the endpoint asked for (the adapter's: Retry-After, else its own backoff); every
+//   call to it meanwhile waits;
+// - when the hold ends, the waiters go in priority order, as many as will go, until the next 429 starts another hold;
+// - the order: the lowest level first, oldest first within a level. A waiter moves up one level for every SCH_AGE_MS

+// a fake clock: sleep(ms, signal) parks a timer (taken off when the signal aborts); advance() moves time timer by timer
+function fakeClock(start = 1_000_000) {
+  let t = start;
+  const timers = new Set();
+  let sleeps = 0;
+  const flush = async () => { for (let i = 0; i < 8; i++) await new Promise((r) => setImmediate(r)); };
+  const sleep = (ms, signal) => new Promise((resolve) => {
+    sleeps++;
+    if (signal && signal.aborted) { resolve(); return; }
+    const tm = { at: t + Math.max(0, ms), resolve, signal, off: null };
+    timers.add(tm);
+    if (signal) { tm.off = () => { timers.delete(tm); resolve(); }; signal.addEventListener('abort', tm.off, { once: true }); }
+  });
+  const fire = (tm) => { timers.delete(tm); if (tm.signal && tm.off) tm.signal.removeEventListener('abort', tm.off); tm.resolve(); };

+    now: () => t,
+    sleep,
+    flush,
+    active: () => timers.size,
+    sleeps: () => sleeps,
+    async advanceTo(target) {
+      await flush();
+      for (let guard = 0; guard < 100000; guard++) {
+        let next = null;
+        for (const tm of timers) if (tm.at <= target && (!next || tm.at < next.at)) next = tm;
+        if (!next) break;
+        t = Math.max(t, next.at);
+        fire(next);
+        await flush();
+      }
+      t = Math.max(t, target);
+      await flush();
+    },
+    advance(ms) { return this.advanceTo(t + ms); },
+    /** advance until done() or `max` ms pass; returns done() */
+    async until(done, max) {
+      const end = t + max;
+      await flush();
+      while (!done() && t < end) {
+        let next = null;
+        for (const tm of timers) if (!next || tm.at < next.at) next = tm;
+        if (!next || next.at > end) { t = end; await flush(); break; }
+        t = Math.max(t, next.at);
+        fire(next);
+        await flush();
+      }
