+/** The remote a push goes to: the repository with the login in it, for that one push only - the stored remote (and
+ * .git/config) never has the token. A folder, ssh or other non-http source as it is. */
+export function setupAuthedRemote(updateSource, creds) {
+  if (!/^https?:/i.test(updateSource)) return updateSource;
+  const u = new URL(updateSource);
+  u.username = creds.user;
+  u.password = creds.token;
+  if (!u.pathname.endsWith('.git')) u.pathname = u.pathname.replace(/\/+$/, '') + '.git';
+  return u.href;

+/** GitLab's tags, without downloading anything: { ok, tags: { name: sha } } or { ok: false, problem }. */
+export function relRemoteTags({ updateSource, creds = {}, cwd = process.cwd() }) {
+  if (!updateSource) return { ok: false, problem: 'The repository is not set.' };
+  const r = rcGit(cwd, ['ls-remote', '--tags', setupAuthedRemote(updateSource, creds)]);
+  if (r.status !== 0) return { ok: false, problem: 'GitLab could not be asked for its tags: ' + rcSaid(r).slice(0, 300) };
+  const tags = {};
+  for (const l of String(r.stdout || '').split(/\r?\n/)) {
+    const m = /^([0-9a-f]{40})\trefs\/tags\/([^\s^]+)$/.exec(l.trim());
+    if (m) tags[m[2]] = m[1];
+  }
+  return { ok: true, tags };
