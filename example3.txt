+function sbiPushLive(repo, args, env) {

+    const c = spawn('git', args, { cwd: repo, env, windowsHide: true, stdio: ['ignore', 'pipe', 'pipe'] });

+    c.on('close', (code) => resolve({ status: code == null ? 1 : code, stdout: out, stderr: err }));
+  });
