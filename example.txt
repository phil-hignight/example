+  if (!lf) return { ok: false, problem: `the committed ${rel} has no line breaks` };
+  if (crlf && crlf !== lf) return { ok: false, problem: `the committed ${rel} mixes CRLF and LF line endings - there is no one kind to match` };

+  if (after === text) return { ok: true, changed: false, eol: name };
