+// streams `script[k]` for the k-th call; a reply with { pauseAt, pauseMs } stops that long at that character
+function adapter(script, seen) {
+  let k = 0;
+  return {
+    async *completeStream(req) {
+      seen.push(req);
+      const r = script[Math.min(k++, script.length - 1)];
+      const text = typeof r === 'string' ? r : r.text;
+      const at = typeof r === 'string' ? -1 : r.pauseAt;
+      for (let i = 0; i < text.length; i += 40) {
+        if (req.signal && req.signal.aborted) return;
+        if (at >= 0 && i <= at && at < i + 40) {
+          yield { choices: [{ delta: { content: text.slice(i, at) } }] };
+          await sleep(r.pauseMs);
+          if (req.signal && req.signal.aborted) return;
+          yield { choices: [{ delta: { content: text.slice(at, i + 40) } }] };
+        } else yield { choices: [{ delta: { content: text.slice(i, i + 40) } }] };
+        await sleep(2);
+      }
+      yield { choices: [{ delta: {}, finish_reason: 'stop' }] };
+    },
+    async complete() { return { choices: [{ message: { content: 'ok' } }] }; },
+  };
